\miniframeson
\begin{frame}[c]{A Prospect-Theoretic Spatial Theory of Vote Buying}
\begin{columns}[T,totalwidth=\textwidth]

% LEFT
\begin{column}{0.55\textwidth}
\begin{itemize}[<+->]
  \item Voters and parties live on a simple \textbf{left--right space}.\\
  {\scriptsize\color{gray}
  Parties \(L,R\) at \(\gamma_L,\gamma_R\); voter \(j\) at \(x_j\).}

  \item The voter prefers the party closer to her ideal point.\\
  {\scriptsize\color{gray}
  \(u_j(\gamma_i)=D-|x_j-\gamma_i|\)}

  \item The voter's preferred party is \(i^\ast\).\\
  {\scriptsize\color{gray}
  \(i^\ast=\arg\max_i u_j(\gamma_i)\)}

  \item The ideological advantage of the preferred party is \(\Delta\).\\
  {\scriptsize\color{gray}
  \(\Delta=u_j(\gamma_{i^\ast})-u_j(\gamma_{-i^\ast})>0\)}
\end{itemize}

\medskip
\onslide<5->{
\begin{block}{Core idea}
{\scriptsize \(\Delta\) is the ideological price of switching sides.}
\end{block}
}
\end{column}

\begin{frame}[c]{Parties Do Not Only Maximize Expected Payoffs}
\begin{columns}[T,totalwidth=\textwidth]

% LEFT
\begin{column}{0.55\textwidth}
\begin{itemize}[<+->]
  \item A party values securing the voter because the vote may protect it from losing.\\
  {\scriptsize\color{gray}
  \(R_i=q(v_i,P_d)W_i\)}

  \item But parties are \textbf{loss averse}: electoral defeat is evaluated as a loss.\\
  {\scriptsize\color{gray}
  \(\lambda_i\geq 1\)}

  \item Loss aversion matters more when electoral loss is salient.\\
  {\scriptsize\color{gray}
  \(\ell_i\in[0,1]\)}

  \item So the party's perceived value of securing the voter is inflated.\\
  {\scriptsize\color{gray}
  \(\widetilde{R}_i=[1+(\lambda_i-1)\ell_i]R_i\)}
\end{itemize}

\medskip
\onslide<5->{
\begin{block}{Effective willingness to pay}
{\scriptsize
\[
\overline{R}_i=\min\{B_i,\widetilde{R}_i\}
\]
}
\end{block}
}

\begin{frame}[c]{Vote Buying: Parties Move First}
\begin{columns}[T,totalwidth=\textwidth]

% LEFT
\begin{column}{0.55\textwidth}
\begin{itemize}[<+->]
  \item Parties choose offers before the voter decides.\\
  {\scriptsize\color{gray}
  Offers \(s_L,s_R\)}

  \item The voter chooses the party giving the highest ideology-plus-transfer payoff.\\
  {\scriptsize\color{gray}
  \(u_j(\gamma_i)+s_i\)}

  \item To buy the vote, party \(i\) must cover the voter's ideological loss.\\
  {\scriptsize\color{gray}
  \(c_i(s_{-i})=\max\{0,u_j(\gamma_{-i})+s_{-i}-u_j(\gamma_i)\}\)}

  \item Buying from a supporter is cheaper than buying from an opponent.
\end{itemize}

\medskip
\onslide<5->{
\begin{block}{Prediction}
{\scriptsize When parties initiate, they mainly target ideologically proximate voters.}
\end{block}
}
\end{column}

\begin{frame}[c]{Vote Buying: Loss Aversion Creates an Insurance Premium}
\begin{columns}[T,totalwidth=\textwidth]

% LEFT
\begin{column}{0.55\textwidth}
\begin{itemize}[<+->]
  \item The minimum offer only makes the voter indifferent.\\
  {\scriptsize\color{gray}
  \(c_i(s_{-i})\)}

  \item But loss-averse parties may pay more to insure against defeat.\\
  {\scriptsize\color{gray}
  \(s_i=c_i(s_{-i})+\rho_i\)}

  \item The insurance premium grows with loss aversion, loss salience, and electoral stakes.\\
  {\scriptsize\color{gray}
  \(\rho_i^\ast=\rho(\lambda_i,\ell_i,R_i)\)}

  \item The premium is feasible only if the party has room to pay it.\\
  {\scriptsize\color{gray}
  \(0\leq \rho_i\leq \overline{R}_i-c_i(s_{-i})\)}
\end{itemize}

\medskip
\onslide<5->{
\begin{block}{Implication}
{\scriptsize Party initiative can raise voter payoffs above the minimum switching price.}
\end{block}
}
\end{column}

  \item The voter names a price for each party.\\
  {\scriptsize\color{gray}

  {\scriptsize\color{gray}

  {\scriptsize\color{gray}

\begin{frame}[c]{Vote Selling: Why Strong but Distant Parties Pay More}
\begin{columns}[T,totalwidth=\textwidth]

% LEFT
\begin{column}{0.55\textwidth}
\begin{itemize}[<+->]
  \item A distant party is costly for the voter to support.\\
  {\scriptsize\color{gray}
  Larger \(\Delta\)}

  \item But an electorally exposed party has more to lose.\\
  {\scriptsize\color{gray}
  Larger \(R_i\), \(\ell_i\), and therefore \(\overline{R}_i\)}

  \item Loss aversion raises the ceiling on acceptable prices.\\
  {\scriptsize\color{gray}
  \(\overline{R}_i=\min\{B_i,[1+(\lambda_i-1)\ell_i]R_i\}\)}

  \item So voters ask more from parties that are both electorally valuable and ideologically costly.
\end{itemize}

\medskip
\onslide<5->{
\begin{block}{Prediction}
{\scriptsize When voters initiate, high prices target electorally strong but ideologically distant parties.}
\end{block}

\miniframeson
\begin{frame}[c]{Formally Derived Hypotheses}
\begin{itemize}[<+->]

  \item[] \textbf{[H1] Core Targeting Under Party Initiative}
  \begin{itemize}
    \item Parties mainly buy from ideologically proximate voters because they are cheaper to secure.\\
    {\scriptsize\color{gray}
    Successful buying requires \(c_i(s_{-i})\leq\overline{R}_i\), and \(c_i\) rises with ideological distance.}
  \end{itemize}

  \medskip

  \item[] \textbf{[H2] Selling to the Opponent Winning Party}
  \begin{itemize}
    \item When voters initiate, they demand higher prices from electorally exposed but ideologically distant parties.\\
    {\scriptsize\color{gray}
    Switching is attractive when
    \(\overline{R}_{-i^\ast}-\overline{R}_{i^\ast}\geq\Delta\).}
  \end{itemize}

  \medskip

  \item[] \textbf{[H3] Higher Voter Payoffs Under Party Initiative}
  \begin{itemize}
    \item Party initiative can give voters higher payoffs because loss-averse parties add an insurance premium.\\
    {\scriptsize\color{gray}
    \(s_i=c_i(s_{-i})+\rho_i\), with \(\rho_i\) increasing in \(\lambda_i\), \(\ell_i\), and \(R_i\).}
  \end{itemize}

\end{itemize}
\end{frame}





\subsection{Hypotheses}


\miniframeson
\begin{frame}[c]{(Formally Derived) Hypotheses}
\begin{itemize}
  \item[] \textbf{[H1] Core Targeting Under Party Initiative:}
  \begin{itemize}
    \item When parties begin, parties mainly buy from their followers.
  \end{itemize}
  \item[]  \textbf{[H2] Selling to the Opponent Winning Party:}
  \begin{itemize}
    \item When voters begin, they demand higher prices from electorally strong {\color{gray}(i.e., with more loss aversion)} but ideologically distant parties.
  \end{itemize}
  \item[]  \textbf{[H3] Higher Voter Payoffs Under Party Initiative:}
  \begin{itemize}
    \item Because parties overspend under electoral risk {\tiny\color{gray} (Bahamonde and Canales, 2022)}, voters make more in vote-buying.
  \end{itemize}
\end{itemize}
\end{frame}

\section{Discussion}

\subsection{What we learned}


\miniframeson
\begin{frame}[c]{Main Takeaways}
\begin{itemize}
  \item {\bf Conceptual move}: integrated {\color{red}vote buying} \emph{and} {\color{blue}vote selling} into the same framework.
  \item {\bf Theory}: formalized a spatial model with one voter and two parties.

    \begin{enumerate}
            \item {\color{red}Parties moved first}.
            \item {\color{blue}Voters moved first}.

    \end{enumerate}


  
  \item {\bf Experiment}: based on the formal model, we designed an econ lab experiment.

  
  \item {\bf Findings}:
  \begin{itemize}
    \item[\checkmark] {\color{red}When parties move first}: transfers go to \textbf{party supporters}.
    \item[\checkmark] {\color{blue}When voters move first}: they \textbf{demand higher prices} from winning parties that are ideologically far away.
    \item[\checkmark] Voters \textbf{make more} when parties initiate the exchange.
  \end{itemize}
\end{itemize}
\end{frame}


\subsection{Why Care}

\begin{frame}[c]{So What? Why This Matters Beyond Clientelism}
\begin{itemize}
  \item \textbf{Who moves first is power.} In many settings the side that moves first \emph{usually} captures more benefits---unions vs. governments or parents vs. the welfare state.
  \medskip
  \item \textbf{Same resources, different winners.} Our paper showed that even with the same agents and setup, simply changing who starts the exchange flips who gets the better deal. This logic also travels to other policy contexts. 
  \medskip
  \item[\checkmark] {\color{blue}If we care about inequality or democracy, we shouldn't just ask \emph{who gets how much}, but also \emph{who moves first}}.
\end{itemize}
\end{frame}
